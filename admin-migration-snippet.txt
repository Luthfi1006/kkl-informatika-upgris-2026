
  <!-- Load Supabase Config & Migration Tools -->
  <script src="assets/js/supabase-config.js"></script>
  <script type="module">
    import { 
      SettingsService, 
      DestinationsService, 
      ProfilesService, 
      GalleryService, 
      VideosService, 
      FeedbackService 
    } from "./lib/supabase-client.js";

    // Migration function
    window.migrateToSupabase = async function() {
      if (!window.__SUPABASE_URL__ || !window.__SUPABASE_KEY__) {
        alert("❌ Supabase credentials belum diload!\n\nPastikan file .env sudah ada dengan credentials yang benar.");
        return;
      }

      if (!confirm("🚀 Migrate data dari JSON ke Supabase?\n\nData existing di Supabase akan ditimpa.\n\nLanjutkan?")) {
        return;
      }

      console.log("🚀 Starting migration...");
      const stats = { destinations: 0, profiles: 0, gallery: 0, videos: 0, feedback: 0, errors: [] };

      try {
        // Load JSON data
        const res = await fetch("./data/kkl_data.json?v=" + Date.now());
        if (!res.ok) throw new Error("Failed to load JSON");
        const jsonData = await res.json();

        // Migrate Settings
        if (jsonData.settings) {
          try {
            await SettingsService.updateSettings({
              site_title: jsonData.settings.site_title,
              tagline: jsonData.settings.tagline,
              dates: jsonData.settings.dates,
              subtext: jsonData.settings.subtext,
              destination_title: jsonData.settings.destination_title,
              countdown_start: jsonData.settings.countdown_start,
              logo_url: jsonData.settings.logo
            });
            console.log("✅ Settings migrated");
          } catch (err) {
            console.error("❌ Settings error:", err);
            stats.errors.push({ type: "settings", error: err.message });
          }
        }

        // Migrate Destinations
        if (jsonData.destinations) {
          for (const dest of jsonData.destinations) {
            try {
              await DestinationsService.create({
                day_no: dest.day_no,
                name: dest.name,
                time: dest.time,
                category: dest.category,
                route: dest.route,
                lat: dest.lat,
                lng: dest.lng,
                description: dest.desc,
                sort_order: dest.id || 0
              });
              stats.destinations++;
            } catch (err) {
              console.error(`❌ Destination ${dest.name}:`, err);
              stats.errors.push({ type: "destination", name: dest.name, error: err.message });
            }
          }
          console.log(`✅ ${stats.destinations} destinations migrated`);
        }

        // Migrate Profiles
        if (jsonData.profiles) {
          for (const prof of jsonData.profiles) {
            try {
              await ProfilesService.create({
                name: prof.name,
                role: prof.role,
                bio: prof.bio,
                photo_url: prof.photo,
                sort_order: prof.sort_order || prof.id || 1
              });
              stats.profiles++;
            } catch (err) {
              console.error(`❌ Profile ${prof.name}:`, err);
              stats.errors.push({ type: "profile", name: prof.name, error: err.message });
            }
          }
          console.log(`✅ ${stats.profiles} profiles migrated`);
        }

        // Migrate Gallery
        if (jsonData.gallery) {
          for (const item of jsonData.gallery) {
            try {
              await GalleryService.create({
                title: item.title,
                category: item.category,
                location: item.location,
                image_url: item.image,
                sort_order: item.id || 0
              });
              stats.gallery++;
            } catch (err) {
              console.error(`❌ Gallery ${item.title}:`, err);
              stats.errors.push({ type: "gallery", title: item.title, error: err.message });
            }
          }
          console.log(`✅ ${stats.gallery} gallery items migrated`);
        }

        // Migrate Videos
        if (jsonData.videos) {
          for (const vid of jsonData.videos) {
            try {
              await VideosService.create({
                title: vid.title,
                youtube_url: vid.embed_url,
                description: vid.desc,
                sort_order: vid.id || 0
              });
              stats.videos++;
            } catch (err) {
              console.error(`❌ Video ${vid.title}:`, err);
              stats.errors.push({ type: "video", title: vid.title, error: err.message });
            }
          }
          console.log(`✅ ${stats.videos} videos migrated`);
        }

        // Migrate Feedback
        if (jsonData.feedback) {
          for (const fb of jsonData.feedback) {
            try {
              await FeedbackService.create({
                name: fb.name,
                rating: fb.rating,
                message: fb.message
              });
              stats.feedback++;
            } catch (err) {
              console.error(`❌ Feedback ${fb.name}:`, err);
              stats.errors.push({ type: "feedback", name: fb.name, error: err.message });
            }
          }
          console.log(`✅ ${stats.feedback} feedback items migrated`);
        }

        console.log("✅ Migration completed!", stats);
        alert(`✅ Migration Berhasil!\n\n` +
          `Destinations: ${stats.destinations}\n` +
          `Profiles: ${stats.profiles}\n` +
          `Gallery: ${stats.gallery}\n` +
          `Videos: ${stats.videos}\n` +
          `Feedback: ${stats.feedback}\n\n` +
          `Errors: ${stats.errors.length}\n\n` +
          `Refresh halaman untuk melihat data dari Supabase!`);

      } catch (err) {
        console.error("❌ Migration failed:", err);
        alert(`❌ Migration gagal: ${err.message}`);
      }
    };

    console.log("💡 Run window.migrateToSupabase() untuk migrate data ke Supabase");
  </script>
